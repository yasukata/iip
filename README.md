# iip: an integratable TCP/IP stack

iip is an integratable TCP/IP stack implementation aiming to allow for easy integration and good performance simultaneously.

**WARNING: The authors will not bear any responsibility if the implementations, provided by the authors, cause any problems.**

## development status

The implementation in this repository is currently undergoing refactoring. The goals are to improve its API and enable formal verification of the entire TCP/IP stack with Frama-C and its Weakest Precondition (WP) plugin.

Many features implemented in earlier versions have not yet been ported. If you need a feature that is not available in the current version, consider using the earlier revision at [d741349](https://github.com/yasukata/iip/tree/d741349ebc5696754988254a911b825ead5d666f).

## other building blocks

### I/O subsystem
- [iip-dpdk](https://github.com/yasukata/iip-dpdk): a DPDK-based backend (Linux); [how to use](https://github.com/yasukata/bench-iip#build).
- [iip-af_xdp](https://github.com/yasukata/iip-af_xdp): an AF_XDP-based backend (Linux); [how to use](https://github.com/yasukata/bench-iip#af_xdp-based-backend).
- [iip-netmap](https://github.com/yasukata/iip-netmap): a netmap-based backend (FreeBSD or Linux); [how to use](https://github.com/yasukata/bench-iip#netmap-based-backend).

### example application
- [bench-iip](https://github.com/yasukata/bench-iip): a benchmark tool.
- [mimicached](https://github.com/yasukata/mimicached): a memcached-compatible cache server.

### simulation
- [iip-ns](https://github.com/yasukata/iip-ns): a helper for running iip on the ns-3 simulator (Linux).

### test
- [ai-test-iip](https://github.com/yasukata/ai-test-iip): a fully AI-generated test suite for iip.

## getting started

Please visit an example application page at [https://github.com/yasukata/bench-iip](https://github.com/yasukata/bench-iip) for testing iip. 

## formal verification with Frama-C

This section describes how to use Frama-C to formally verify iip. The current focus is on the runtime error (RTE) checks generated by the RTE plugin activated with the ```-wp-rte``` option. These checks help identify potential issues such as invalid memory access and arithmetic overflow.

Please make a directory named ```frama-c``` and enter it with the following commands.

```
mkdir frama-c
```

```
cd frama-c
```

Please save the following C programs as ```iosub.c``` and ```main.c``` in the ```frama-c``` directory.

#ifndef SKIP_IOSUB_DEFINITION

#define NUM_PKT (8192)
#define PKT_BUF_SIZE (1024)

struct iosub_info {
	uint8_t ethernet_addr[6];
	uint32_t ipv4_addr_be;
	uint8_t pkt_buf[NUM_PKT][2 /* len */ + 2 /* next */ + PKT_BUF_SIZE];
};

typedef struct iosub_info * opaque_ptr_t;
#define IIP_OPAQUE_P opaque_ptr_t
#define IIP_PKT_P uint16_t

#endif

#ifndef SKIP_IOSUB_IMPLEMENTATION

#define ETH_ADDR_LEN 6
#define ETH_HDR_LEN 14

/*@ predicate f_iip_ops_pkt_valid(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque) = pkt_ref < NUM_PKT ? true : false; */
/*@
	assigns \nothing;
	ensures \result ==> f_iip_ops_pkt_valid(pkt_ref, opaque);
 */
static bool iip_ops_pkt_valid(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	if (pkt_ref < NUM_PKT)
		return true;
	else
		return false;
}
#define IIP_OPS_PKT_VALID() do { iip_ret_bool = iip_ops_pkt_valid(pkt, opaque); } while (0)

/*@ logic uint8_t * f_iip_ops_pkt_get_data{L}(integer pkt_ref, IIP_OPAQUE_P opaque) = &opaque->pkt_buf[pkt_ref][2]; */
/*@ logic integer f_iip_ops_pkt_get_len{L}(integer pkt_ref, IIP_OPAQUE_P opaque) = ((uint16_t) opaque->pkt_buf[pkt_ref][0] * 256) + opaque->pkt_buf[pkt_ref][1]; */
/*@ logic integer f_iip_ops_pkt_get_capacity{L}(integer pkt_ref, IIP_OPAQUE_P opaque) = PKT_BUF_SIZE; */

/*@
	requires \valid_read(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns \result \from opaque, pkt_ref;
	ensures \result == f_iip_ops_pkt_get_data(pkt_ref, opaque);
 */
static uint8_t *iip_ops_pkt_get_data(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	return &opaque->pkt_buf[pkt_ref][4];
}
#define IIP_OPS_PKT_GET_DATA() do { iip_ret_u8_ptr = iip_ops_pkt_get_data(pkt, opaque); } while (0)

/*@
	requires \valid_read(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns \nothing;
	ensures \result == f_iip_ops_pkt_get_len(pkt_ref, opaque);
 */
static IIP_PKT_LEN_T iip_ops_pkt_get_len(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	return ((uint16_t) opaque->pkt_buf[pkt_ref][0] * 256) + opaque->pkt_buf[pkt_ref][1];
}
#define IIP_OPS_PKT_GET_LEN() do { iip_ret_len = iip_ops_pkt_get_len(pkt, opaque); } while (0)

/*@
	requires \valid(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns opaque->pkt_buf[pkt_ref][0 .. 1] \from len;
	ensures (opaque->pkt_buf[pkt_ref][0] == len / 256) && (opaque->pkt_buf[pkt_ref][1] == len % 256);
 */
static int iip_ops_pkt_set_len(IIP_PKT_P pkt_ref, uint16_t len, IIP_OPAQUE_P opaque)
{
	opaque->pkt_buf[pkt_ref][0] = len / 256;
	opaque->pkt_buf[pkt_ref][1] = len % 256;
	return 0;
}
#define IIP_OPS_PKT_SET_LEN() do { iip_ret_int = iip_ops_pkt_set_len(pkt, len, opaque); } while (0)

/*@
	requires \valid_read(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns \nothing;
	ensures \result == f_iip_ops_pkt_get_capacity(pkt_ref, opaque);
 */
static uint16_t iip_ops_pkt_get_capacity(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	return PKT_BUF_SIZE;
}
#define IIP_OPS_PKT_GET_CAPACITY() do { iip_ret_len = iip_ops_pkt_get_capacity(pkt, opaque); } while (0)

/*@
	requires \valid(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns \nothing;
 */
static IIP_PKT_P iip_ops_pkt_scatter_gather_chain_get_next(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	return ((uint16_t) opaque->pkt_buf[pkt_ref][2] * 256) + opaque->pkt_buf[pkt_ref][3];
}
#define IIP_OPS_PKT_SCATTER_GATHER_CHAIN_GET_NEXT() do { iip_ret_pkt = iip_ops_pkt_scatter_gather_chain_get_next(head_pkt, tail_pkt, opaque); } while (0)

/*@
	requires \valid(opaque);
	requires f_iip_ops_pkt_valid(head_pkt_ref, opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns opaque->pkt_buf[0 .. NUM_PKT - 1][2 .. 3] \from pkt_ref;
 */
static int iip_ops_pkt_scatter_gather_chain_append(IIP_PKT_P head_pkt_ref, IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	IIP_PKT_P ref = head_pkt_ref, i;
	/*@
		loop invariant 0 <= i <= NUM_PKT;
		loop invariant f_iip_ops_pkt_valid(ref, opaque);
		loop assigns i, ref, opaque->pkt_buf[0 .. NUM_PKT - 1][2 .. 3];
		loop variant NUM_PKT - i;
	 */
	for (i = 0; i < NUM_PKT; i++) {
		IIP_PKT_P next = iip_ops_pkt_scatter_gather_chain_get_next(ref, opaque);
		if (!iip_ops_pkt_valid(next, opaque)) {
			opaque->pkt_buf[ref][2] = pkt_ref / 256;
			opaque->pkt_buf[ref][3] = pkt_ref % 256;
			break;
		}
		ref = next;
	}
	return 0;
}
#define IIP_OPS_PKT_SCATTER_GATHER_CHAIN_APPEND() do { iip_ret_int = iip_ops_pkt_scatter_gather_chain_append(head_pkt, pkt, opaque); } while (0)

/*@
	requires \valid(opaque);
	requires \valid(pkt_ref);
	assigns *pkt_ref, opaque->pkt_buf[0 .. NUM_PKT - 1][0 .. 1];
	ensures \result == 0 ==> f_iip_ops_pkt_valid(*pkt_ref, opaque);
 */
static int iip_ops_pkt_alloc(IIP_PKT_P *pkt_ref, IIP_OPAQUE_P opaque)
{
	uint16_t i;
	/*@
		loop invariant 0 <= i <= NUM_PKT;
		loop assigns i;
		loop variant NUM_PKT - i;
	 */
	for (i = 0; i < NUM_PKT; i++) {
		if (!iip_ops_pkt_get_len(i, opaque))
			break;
	}
	if (i < NUM_PKT) {
		opaque->pkt_buf[i][0] = 0xff;
		opaque->pkt_buf[i][1] = 0xff;
		*pkt_ref = i;
		return 0;
	} else
		return -1;
}
#define IIP_OPS_PKT_ALLOC() do { iip_ret_int = iip_ops_pkt_alloc(pkt, opaque); } while (0)

/*@
	requires \valid(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns opaque->pkt_buf[pkt_ref][0 .. 1];
	ensures \result == 0 ==> opaque->pkt_buf[pkt_ref][0] == 0;
	ensures \result == 0 ==> opaque->pkt_buf[pkt_ref][1] == 0;
 */
static int iip_ops_pkt_free(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	if (!opaque->pkt_buf[pkt_ref][0] && !opaque->pkt_buf[pkt_ref][1])
		return -1;
	opaque->pkt_buf[pkt_ref][0] = 0;
	opaque->pkt_buf[pkt_ref][1] = 0;
	return 0;

}
#define IIP_OPS_PKT_FREE() do { iip_ret_int = iip_ops_pkt_free(pkt, opaque); } while (0)

/*@
	requires \valid(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	requires cloned_pkt < NUM_PKT;
	requires cloned_pkt != pkt_ref;
	requires f_iip_ops_pkt_get_len(pkt_ref, opaque) <= PKT_BUF_SIZE;
	assigns opaque->pkt_buf[cloned_pkt][0 .. PKT_BUF_SIZE + 1] \from opaque->pkt_buf[pkt_ref][0 .. PKT_BUF_SIZE + 1];
	ensures \forall integer j; 0 <= j < 2 + f_iip_ops_pkt_get_len(pkt_ref, opaque) ==>
				opaque->pkt_buf[cloned_pkt][j] == opaque->pkt_buf[pkt_ref][j];
	ensures \forall integer j; 0 <= j < f_iip_ops_pkt_get_len(pkt_ref, opaque) ==>
				f_iip_ops_pkt_get_data(cloned_pkt, opaque)[j] == f_iip_ops_pkt_get_data(pkt_ref, opaque)[j];
	ensures f_iip_ops_pkt_get_len(pkt_ref, opaque) == f_iip_ops_pkt_get_len(cloned_pkt, opaque);
 */
static void iip_ops_pkt_clone__copy_data(IIP_PKT_P pkt_ref, IIP_PKT_P cloned_pkt, IIP_OPAQUE_P opaque)
{
		uint16_t i;
		/*@
			loop invariant 0 <= i <= 2 + f_iip_ops_pkt_get_len(pkt_ref, opaque);
			loop invariant \forall integer j; 0 <= j < i <= 2 + f_iip_ops_pkt_get_len(pkt_ref, opaque) ==>
				opaque->pkt_buf[cloned_pkt][j] == opaque->pkt_buf[pkt_ref][j];
			loop assigns i, opaque->pkt_buf[cloned_pkt][0 .. PKT_BUF_SIZE + 1] \from opaque->pkt_buf[pkt_ref][0 .. PKT_BUF_SIZE + 1];
			loop variant 2 + f_iip_ops_pkt_get_len(pkt_ref, opaque) - i;
		 */
		for (i = 0; i < 2 + iip_ops_pkt_get_len(pkt_ref, opaque); i++)
			opaque->pkt_buf[cloned_pkt][i] = opaque->pkt_buf[pkt_ref][i];
		/*@ assert opaque->pkt_buf[cloned_pkt][0] == opaque->pkt_buf[pkt_ref][0]
			&& opaque->pkt_buf[cloned_pkt][1] == opaque->pkt_buf[pkt_ref][1]; */
		/*@ assert f_iip_ops_pkt_get_len(pkt_ref, opaque) == f_iip_ops_pkt_get_len(cloned_pkt, opaque); */
}

/*@
	requires \valid(opaque);
	requires \valid(cloned_pkt);
	requires \separated(cloned_pkt, opaque);
	assigns *cloned_pkt, *opaque;
	ensures \result == 0 ==> *cloned_pkt < NUM_PKT;
	ensures \result == 0 ==> f_iip_ops_pkt_get_len(*cloned_pkt, opaque) == f_iip_ops_pkt_get_len(pkt_ref, opaque);
	ensures \result == 0 ==> \forall integer i; 0 <= i < f_iip_ops_pkt_get_len(*cloned_pkt, opaque) ==>
		f_iip_ops_pkt_get_data(*cloned_pkt, opaque)[i] == f_iip_ops_pkt_get_data(pkt_ref, opaque)[i];
 */
static int iip_ops_pkt_clone(IIP_PKT_P pkt_ref, IIP_PKT_P *cloned_pkt, IIP_OPAQUE_P opaque)
{
	if (!iip_ops_pkt_valid(pkt_ref, opaque))
		return -1;
	{
		IIP_PKT_P new_pkt;
		if (iip_ops_pkt_alloc(&new_pkt, opaque))
			return -1;
		if (new_pkt == pkt_ref)
			return -1;
		if (iip_ops_pkt_get_len(pkt_ref, opaque) > iip_ops_pkt_get_capacity(pkt_ref, opaque)) {
			if (iip_ops_pkt_free(new_pkt, opaque))
				return -1;
			return -1;
		}
		iip_ops_pkt_clone__copy_data(pkt_ref, new_pkt, opaque);
		*cloned_pkt = new_pkt;
		return 0;
	}
}
#define IIP_OPS_PKT_CLONE() do { iip_ret_int = iip_ops_pkt_clone(pkt, cloned_pkt, opaque); } while (0)

/*@
	requires \valid_read(opaque);
	requires \valid_read(ethernet_addr + (0 .. ETH_ADDR_LEN - 1));
	assigns \nothing;
	ensures \result == 0 ==> \forall integer j; 0 <= j < ETH_ADDR_LEN ==> opaque->ethernet_addr[j] == ethernet_addr[j];
 */
static int iip_ops_ethernet_addr_match(const uint8_t ethernet_addr[ETH_ADDR_LEN], IIP_OPAQUE_P opaque)
{
	uint16_t i;
	/*@
		loop invariant 0 <= i <= ETH_ADDR_LEN;
		loop invariant \forall integer j; 0 <= j < i <= ETH_ADDR_LEN ==> opaque->ethernet_addr[j] == ethernet_addr[j];
		loop assigns i;
		loop variant ETH_ADDR_LEN - i;
	 */
	for (i = 0; i < ETH_ADDR_LEN; i++) {
		if (opaque->ethernet_addr[i] != ethernet_addr[i])
			return -1;
	}
	return 0;
}
#define IIP_OPS_ETHERNET_ADDR_MATCH() do { iip_ret_bool = iip_ops_ethernet_addr_match(buf, opaque); } while (0)

/*@
	requires \valid_read(opaque);
	requires \valid(buf + (0 .. ETH_HDR_LEN - 1));
	requires \valid_read(dst_ethernet_addr + (0 .. 5));
	assigns buf[ 0 ..  5] \from dst_ethernet_addr[0 .. 5];
	assigns buf[ 6 .. 11] \from opaque->ethernet_addr[0 .. 5];
	assigns buf[12 .. 13] \from proto_be;
 */
static int iip_ops_ethernet_hdr_craft(uint8_t *buf, const uint8_t dst_ethernet_addr[ETH_ADDR_LEN], uint16_t proto_be, IIP_OPAQUE_P opaque)
{
	buf[ 0] = dst_ethernet_addr[0];
	buf[ 1] = dst_ethernet_addr[1];
	buf[ 2] = dst_ethernet_addr[2];
	buf[ 3] = dst_ethernet_addr[3];
	buf[ 4] = dst_ethernet_addr[4];
	buf[ 5] = dst_ethernet_addr[5];
	buf[ 6] = opaque->ethernet_addr[0];
	buf[ 7] = opaque->ethernet_addr[1];
	buf[ 8] = opaque->ethernet_addr[2];
	buf[ 9] = opaque->ethernet_addr[3];
	buf[10] = opaque->ethernet_addr[4];
	buf[11] = opaque->ethernet_addr[5];
	buf[12] = proto_be % 256;
	buf[13] = proto_be / 256;
	return 0;
}
#define IIP_OPS_ETHERNET_HDR_CRAFT() do { iip_ret_int = iip_ops_ethernet_hdr_craft(buf, dst_addr, proto_be, opaque); } while (0)

/*@
	requires \valid(opaque);
	requires f_iip_ops_pkt_valid(pkt_ref, opaque);
	assigns *opaque;
 */
static int iip_ops_ethernet_push(IIP_PKT_P pkt_ref, IIP_OPAQUE_P opaque)
{
	if (pkt_ref % 2)
		return 1;
	{
		IIP_PKT_P ref = pkt_ref, i;
		/*@
			loop invariant 0 <= i <= NUM_PKT;
			loop assigns i, ref, *opaque;
			loop variant NUM_PKT - i;
		 */
		for (i = 0; i < NUM_PKT; i++) {
			if (iip_ops_pkt_valid(ref, opaque)) {
				IIP_PKT_P next = iip_ops_pkt_scatter_gather_chain_get_next(ref, opaque);
				iip_ops_pkt_free(ref, opaque);
				ref = next;
			} else
				break;
		}
	}
	return 0;
}
#define IIP_OPS_ETHERNET_PUSH() do { iip_ret_int = iip_ops_ethernet_push(pkt, opaque); } while (0)

/*@
	requires \valid(opaque);
	assigns \nothing;
 */
static int iip_ops_ethernet_flush(IIP_OPAQUE_P opaque)
{
	if (opaque->ipv4_addr_be % 2)
		return 1;
	return 0;
}
#define IIP_OPS_ETHERNET_FLUSH() do { iip_ret_int = iip_ops_ethernet_flush(opaque); } while (0)

/*@
	requires \valid_read(opaque);
	assigns \nothing;
	ensures \result == 0 ==> opaque->ipv4_addr_be == ipv4_addr_be;
 */
static int iip_ops_ipv4_addr_match(uint32_t ipv4_addr_be, IIP_OPAQUE_P opaque)
{
	if (opaque->ipv4_addr_be == ipv4_addr_be)
		return 0;
	else
		return -1;
}
#define IIP_OPS_IPV4_ADDR_MATCH() do { iip_ret_bool = iip_ops_ipv4_addr_match(ii_read_uint32(ii_call_pkt_get_data(rx_pkt, opaque) + II_ETH_HDR_LEN + II_ARP_HDR_LEN + 16), opaque); } while (0)

#undef NUM_PKT
#undef PKT_BUF_SIZE
#undef ETH_ADDR_LEN
#undef ETH_HDR_LEN

#endif

The following command attempts verification for the entire iip TCP/IP stack implementation. This may take several hours. Note that the ```-wp-print``` option shows many details, therefore, we recommend redirecting it to a file.

```
frama-c -cpp-extra-args=-D_ENABLE_FRAMA_C_CHECK,-CC -wp-prover alt-ergo,cvc4,cvc5,z3 -wp -wp-rte -wp-smoke-tests -wp-print main.c
```

We can attempt the verification for a particular function specified through the ```-wp-fct``` option. This may complete in a few seconds. The following is an example command that performs the verification only for ```iip_run``` (without ```-wp-print```).

```
frama-c -cpp-extra-args=-D_ENABLE_FRAMA_C_CHECK,-CC -wp-prover alt-ergo,cvc4,cvc5,z3 -wp -wp-rte -wp-smoke-tests main.c -wp-fct iip_run
```
